class BidirectionalInferenceNode:
    """Manages concurrent dual-direction vector mapping and inverse reconstructions."""
    def __init__(self, resonance_base: float = 7.83) -> None:
        self.scale_factor = math.sin(resonance_base)

    def forward_pass(self, features: Tuple[float, ...]) -> Tuple[float, ...]:
        return tuple(round(x * self.scale_factor, 6) for x in features)

    def reverse_pass(self, latents: Tuple[float, ...]) -> Tuple[float, ...]:
        return tuple(round(y / self.scale_factor, 6) for y in latents)


class LoopbackConvergenceEngine:
    """Routes node outputs directly into incoming input buffers recursively."""
    def __init__(self, node: BidirectionalInferenceNode, tolerance: float = 1e-5) -> None:
        self.node = node
        self.tolerance = tolerance

    def iterate_loopback(self, initial_vector: Tuple[float, ...], max_depth: int = 3) -> Dict[str, Any]:
        current_vector = initial_vector
        variance = 1.0
        cycles = 0
        
        while cycles < max_depth and variance > self.tolerance:
            cycles += 1
            fwd = self.node.forward_pass(current_vector)
            rev = self.node.reverse_pass(fwd)
            variance = sum(abs(o - r) for o, r in zip(current_vector, rev))
            current_vector = rev

        return {
            "TOTAL_CYCLES": cycles, 
            "FINAL_LOSS_VARIANCE": round(variance, 8), 
            "TERMINAL_VECTOR": current_vector
        }
