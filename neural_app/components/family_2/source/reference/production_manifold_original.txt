        return tuple(round(y / self.scale_factor, 6) for y in latents)


class LoopbackConvergenceEngine:
    """Routes node outputs directly into incoming input buffers recursively."""
    def __init__(self, node: BidirectionalInferenceNode, tolerance: float = 1e-5) -> None:
        self.node = node
        self.tolerance = tolerance

    def iterate_loopback(self, initial_vector: Tuple[float, ...], max_depth: int = 3) -> Dict[str, Any]:
        current_vector = initial_vector
        variance = 1.0
        cycles = 0
        
        while cycles < max_depth and variance > self.tolerance:
            cycles += 1

            "TOTAL_CYCLES": cycles, 
            "FINAL_LOSS_VARIANCE": round(variance, 8), 
            "TERMINAL_VECTOR": current_vector
        }

class TernaryConverter:
    """Converts standard numerical representations into base-3 radix structures."""
    @staticmethod
    def decimal_to_ternary(n: int) -> str:
        if n == 0:
            return "0"
        digits = []
        while n > 0:
            digits.append(str(n % 3))
            n //= 3
        return "".join(reversed(digits))

    @staticmethod
    def hex_to_ternary(hex_str: str) -> str:
        try:
            clean_hex = "".join(c for c in hex_str if c.isalnum())
            # Safely handle mixed strings or pure hex hashes
            decimal_val = int(clean_hex, 16)
        except ValueError:
            # Fallback to string-hash processing if alphanumeric token is not pure hex
            decimal_val = abs(hash(hex_str))
        return TernaryConverter.decimal_to_ternary(decimal_val)
