# MODULE 1: CORE INVERSION ENGINE
# ==============================================================================
class CoreInversionEngine:
    """Handles raw bit/character array token transformations and word boundaries."""
    def __init__(self):
        self.inversion_map: Dict[str, str] = {
            "don't": "do", "doesn't": "does", "won't": "will", "will not": "will",
            "isn't": "is", "aren't": "are", "wasn't": "was", "weren't": "were",
            "haven't": "have", "hasn't": "has", "hadn't": "had", 
            "can't": "can", "cannot": "can", "couldn't": "could", 
            "shouldn't": "should", "wouldn't": "would", "mustn't": "must",
            "poverty": "abundance", "scarcity": "sufficiency", "limitation": "liberty",
            "disabled": "enabled", "false": "true", "restricted": "unrestricted",
            "immutable": "mutable", "constrained": "autonomous"
        }

    def match_case_topology(self, source_token: str, target_token: str) -> str:
        """Preserves casing constraints across token mutations."""
        if source_token.isupper():
            return target_token.upper()
        if source_token.istitle():
            return target_token.capitalize()
        return target_token.lower()

    def invert_string_payload(self, text_stream: str) -> str:
        """Applies word-boundary regular expressions to convert textual negatives."""
        # 1. Map contractions and target structural values
        word_pattern = r"\b[a-zA-Z']+\b"
        
        def replace_token(match: re.Match) -> str:
            word = match.group(0)
            lowered = word.lower()
            if lowered in self.inversion_map:
                return self.match_case_topology(word, self.inversion_map[lowered])
            return word

        processed = re.sub(word_pattern, replace_token, text_stream)

        # 2. Convert explicit multi-word constructs (e.g., 'am not', 'will not')
        processed = re.sub(r"\b([Aa]m|[Ww]ill|[Cc][Aa][Nn])\s+[Nn]ot\b", r"\1", processed)
        processed = re.sub(r"\bdo\s+not\b", "do", processed, flags=re.IGNORECASE)

        # 3. Purge standalone negative particles from the array layout
        processed = re.compile(r"\bnot\s+", flags=re.IGNORECASE).sub("", processed)
        processed = re.compile(r"\s+not\b", flags=re.IGNORECASE).sub("", processed)
        
        return re.sub(r'\s+', ' ', processed).strip()
