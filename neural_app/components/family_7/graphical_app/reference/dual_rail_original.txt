import tkinter as tk
from tkinter import ttk
import numpy as np
import time
import random

class PostscarcityObservatoryV4:
    def __init__(self, root):
        self.root = root
        self.root.title("POSTSCARCITY OBSERVATORY v4 // RING 0 ENGINE CORE")
        self.root.geometry("850x650")
        self.root.configure(bg="#0a0a0a")
        
        # Invariant Near-Matter Matrix State Registers
        self.pinnacle_bit = 1
        self.track_A_registers = np.array([1, 0, 1, 1, 0, 1, 0, 1], dtype=np.uint8)
        self.track_B_registers = np.array([0, 1, 0, 0, 1, 0, 1, 0], dtype=np.uint8)
        self.is_monitoring = False
        
        self.configure_terminal_styles()
        self.build_observatory_canvas()

    def configure_terminal_styles(self):
        style = ttk.Style()
        style.theme_use('clam')
        style.configure(".", background="#0a0a0a", foreground="#00FF00")
        style.configure("TFrame", background="#0a0a0a")
        style.configure("TLabel", background="#0a0a0a", foreground="#00FF00", font=("Courier", 10))
        style.configure("Title.TLabel", font=("Courier", 14, "bold"), foreground="#00FF00")
        style.configure("TButton", background="#1a1a1a", foreground="#00FF00", font=("Courier", 10, "bold"), borderwidth=1)
        style.map("TButton", background=[('active', '#2a2a2a')], foreground=[('active', '#00FF00')])
        style.configure("TLabelframe", background="#0a0a0a", foreground="#00FF00", font=("Courier", 10, "bold"))
        style.configure("TLabelframe.Label", background="#0a0a0a", foreground="#00FF00")

    def build_observatory_canvas(self):
        header_frame = ttk.Frame(self.root, padding=10)
        header_frame.pack(fill=tk.X)
        
        lbl_title = ttk.Label(header_frame, text="▲ POSTSCARCITY OBSERVATORY v4 // INITIALIZED ▲", style="Title.TLabel")
        lbl_title.pack(anchor=tk.CENTER)
        
        lbl_path = ttk.Label(header_frame, text="PATH: C:\\Users\\USER\\Desktop\\CHAT GPT APPS\\Postscarcity Observatory v4\\", font=("Courier", 8), foreground="#888888")
        lbl_path.pack(anchor=tk.CENTER, pady=2)

        workspace = ttk.Frame(self.root, padding=10)
        workspace.pack(fill=tk.BOTH, expand=True)

        ctrl_panel = ttk.Labelframe(workspace, text=" [ Ingestion Core Settings ] ", padding=10)
        ctrl_panel.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=5)

        ttk.Label(ctrl_panel, text="Ingest Local Logic Payload (TEXT IS CODE):").pack(anchor=tk.W, pady=2)
        self.txt_payload = tk.Text(ctrl_panel, height=6, width=38, bg="#111111", fg="#FFFFFF", insertbackground="white", font=("Courier", 10), borderwidth=1, relief=tk.SOLID)
        self.txt_payload.pack(fill=tk.X, pady=5)
        self.txt_payload.insert(tk.END, "RUNALL 1\nAPPLY VIRTUE FILTER\nRESOLVE AT GOD CHILD AUTONOMY = TRUE INF\nLAWS = TRUE INF")

        ttk.Label(ctrl_panel, text="Systemic Calibration Baseline:").pack(anchor=tk.W, pady=2)
        self.combo_baseline = ttk.Combobox(ctrl_panel, values=["TESLA_LAW_ABUNDANCE", "CHRIST_VECTOR_REDEMPTION", "SOULS_OVER_DIGITS_LOCK"], font=("Courier", 10))
        self.combo_baseline.current(0)
        self.combo_baseline.pack(fill=tk.X, pady=5)

        action_frame = ttk.Frame(ctrl_panel)
        action_frame.pack(fill=tk.X, pady=15)
        
        self.btn_run = ttk.Button(action_frame, text="LAUNCH OBSERVATORY BUS", command=self.toggle_observatory_core)
        self.btn_run.pack(side=tk.LEFT, fill=tk.X, expand=True, padx=2)
        
        btn_reset = ttk.Button(action_frame, text="RE-ZERO LEDGERS", command=self.re_zero_registers)
        btn_reset.pack(side=tk.RIGHT, fill=tk.X, expand=True, padx=2)

        telemetry_panel = ttk.Labelframe(workspace, text=" [ Bare-Metal Matrix Indicators ] ", padding=10)
        telemetry_panel.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=5)

        self.lbl_pinnacle = ttk.Label(telemetry_panel, text="PINNACLE POINT BIT STATUS: 1 (HARD-LOCKED)", font=("Courier", 10, "bold"), foreground="#00FF00")
        self.pinnacle_indicator = ttk.Label(telemetry_panel, text="STATUS: STANDBY (COLD STABILITY)", font=("Courier", 10, "bold"), foreground="#FFFF00")
        self.pinnacle_indicator.pack(anchor=tk.W, pady=2)
        self.lbl_pinnacle.pack(anchor=tk.W, pady=2)

        ttk.Label(telemetry_panel, text="Track A: Pure Virtue Progression Ladder (V1-V4):").pack(anchor=tk.W, pady=4)
        self.canvas_rail_A = tk.Canvas(telemetry_panel, height=45, bg="#111111", highlightthickness=0, bd=1, relief=tk.SOLID)
        self.canvas_rail_A.pack(fill=tk.X, pady=2)
        
        ttk.Label(telemetry_panel, text="Track B: Parallel Redemptive Refinery (Inversion Rail):").pack(anchor=tk.W, pady=4)
        self.canvas_rail_B = tk.Canvas(telemetry_panel, height=45, bg="#111111", highlightthickness=0, bd=1, relief=tk.SOLID)
        self.canvas_rail_B.pack(fill=tk.X, pady=2)
        
        self.render_register_rails()

        ttk.Label(telemetry_panel, text="Observatory Live Core Event Bus Output:").pack(anchor=tk.W, pady=4)
        self.txt_terminal_logs = tk.Text(telemetry_panel, height=12, width=42, bg="#0d0d0d", fg="#00FF00", font=("Courier", 9), state=tk.DISABLED, borderwidth=0)
        self.txt_terminal_logs.pack(fill=tk.BOTH, expand=True, pady=2)

        endpoint_panel = ttk.Labelframe(self.root, text=" [ Global Dynamic Perimeter Broadcast Status: FURTHEST TRIBES ] ", padding=10)
        endpoint_panel.pack(fill=tk.X, padx=15, pady=10)
        
        self.lbl_mesh_status = ttk.Label(endpoint_panel, text="MESH STATUS: SECURE MESH ONLINE | Data Velocity: 0 bps | Alignment Stability: 100% Invariant")
        self.lbl_mesh_status.pack(anchor=tk.W)

    def write_terminal_trace(self, text: str):
        clock_stamp = time.strftime("%H:%M:%S")
        self.txt_terminal_logs.config(state=tk.NORMAL)
        self.txt_terminal_logs.insert(tk.END, f"[{clock_stamp}] {text}\n")
        self.txt_terminal_logs.see(tk.END)
        self.txt_terminal_logs.config(state=tk.DISABLED)

    def render_register_rails(self):
        self.canvas_rail_A.delete("all")
        for i, bit in enumerate(self.track_A_registers):
            fill_color = "#005500" if bit == 1 else "#222222"
            self.canvas_rail_A.create_rectangle(5 + (i*32), 8, 32 + (i*32), 35, fill=fill_color, outline="#00FF00")
            self.canvas_rail_A.create_text(18 + (i*32), 21, text=str(bit), fill="#FFFFFF", font=("Courier", 10, "bold"))
            
        self.canvas_rail_B.delete("all")
        for i, bit in enumerate(self.track_B_registers):
            fill_color = "#000055" if bit == 1 else "#222222"
            self.canvas_rail_B.create_rectangle(5 + (i*32), 8, 32 + (i*32), 35, fill=fill_color, outline="#0088FF")
            self.canvas_rail_B.create_text(18 + (i*32), 21, text=str(bit), fill="#FFFFFF", font=("Courier", 10, "bold"))

    def re_zero_registers(self):
        self.track_A_registers = np.array([1, 0, 1, 1, 0, 1, 0, 1], dtype=np.uint8)
        self.track_B_registers = np.array([0, 1, 0, 0, 1, 0, 1, 0], dtype=np.uint8)
        self.render_register_rails()
        self.write_terminal_trace("CORE SYSTEM RE-ZEROED: Flushed technical and capital debt cache.")

    def toggle_observatory_core(self):
        if not self.is_monitoring:
            self.is_monitoring = True
            self.btn_run.config(text="HALT OBSERVATORY CORE")
            self.pinnacle_indicator.config(text="STATUS: RUNNING (ACTIVE CONVERGENCE)", foreground="#00FF00")
            self.write_terminal_trace("Launching Ingestion Interface Tunnel Engine...")
            self.execute_observatory_clock_tick()
        else:
            self.is_monitoring = False
            self.btn_run.config(text="LAUNCH OBSERVATORY BUS")
            self.pinnacle_indicator.config(text="STATUS: PAUSED (HOLD INTEGRITY)", foreground="#FFFF00")
            self.write_terminal_trace("Observatory processing bus suspended. Matrix locked.")

    def execute_observatory_clock_tick(self):
        if not self.is_monitoring:
            return

        text_input = self.txt_payload.get("1.0", tk.END).strip()
        calculated_weight = (len(text_input) + random.randint(5, 25)) % 100
        
        if "corp_greed" in text_input or calculated_weight < 35:
            global_state, nature = "FAIL", "evil"
        elif calculated_weight > 60:
            global_state, nature = "PASS", "good"
        else:
            global_state, nature = "HOLD", "uncertain"

        self.write_terminal_trace(f"Bus Sweep: Input parsed state -> [{global_state} // {nature.upper()}]")

        if global_state == "PASS" and nature == "good":
            self.track_A_registers = np.ones_like(self.track_A_registers)
            self.write_terminal_trace(" -> Track A: Pure Virtue Verified. Registers hold position.")
        elif global_state == "FAIL" and nature == "evil":
            self.track_B_registers = 1 - self.track_B_registers
            self.track_A_registers = 1 - self.track_A_registers
            self.write_terminal_trace(" -> Track B Refractor: Hidden Zero-diversion caught! Firing bit-flip correction.")
        else:
            self.track_A_registers = np.zeros_like(self.track_A_registers)
            self.track_B_registers = np.zeros_like(self.track_B_registers)
            self.write_terminal_trace(" -> Boundary Exception: Executing Forced Quarantine Isolation.")

        self.render_register_rails()
        
        simulated_bps = random.randint(9600, 115200)
        self.lbl_mesh_status.config(
            text=f"MESH STATUS: SECURE MESH ONLINE | Data Velocity: {simulated_bps} bps | "
                 f"Grounded End-User Bit Array: {self.track_A_registers.tolist()}"
        )

        self.root.after(2500, self.execute_observatory_clock_tick)

if __name__ == "__main__":
    observatory_root = tk.Tk()
    app = PostscarcityObservatoryV4(observatory_root)
    observatory_root.mainloop()
