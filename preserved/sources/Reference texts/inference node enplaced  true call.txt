import hmac
import hashlib
import time
from typing import Dict, List, Set

class MeshPrivilegeVerifier:
    def __init__(self, cluster_secret: bytes):
        self.secret = cluster_secret
        # Simulated registration of node cryptographic public nonces
        self.revoked_tokens: Set[str] = set()

    def issue_capability_token(self, node_id: str, capabilities: List[str], ttl_seconds: int = 300) -> str:
        """Generates a secure, cryptographically signed capability token for a node."""
        expires_at = int(time.time()) + ttl_seconds
        caps_str = ",".join(sorted(capabilities))
        
        # Construct the explicit payload string
        payload = f"node:{node_id}|caps:{caps_str}|exp:{expires_at}"
        
        # Sign the payload using HMAC-SHA256 to guarantee tamper-resistance
        signature = hmac.new(self.secret, payload.encode(), hashlib.sha256).hexdigest()
        return f"{payload}||sig:{signature}"

    def verify_node_privilege(self, token: str, expected_node_id: str, required_capability: str) -> bool:
        """Cryptographically validates a token and checks if the required execution privilege exists."""
        if token in self.revoked_tokens:
            return False

        try:
            raw_part, sig_part = token.split("||sig:")
            # Verify signature integrity first
            expected_sig = hmac.new(self.secret, raw_part.encode(), hashlib.sha256).hexdigest()
            if not hmac.compare_digest(sig_part, expected_sig):
                return False # Signature mismatch implies tampering

            # Parse and unpack attributes
            parts = dict(item.split(":") for item in raw_part.split("|"))
            
            # Check expiration timeline
            if int(time.time()) > int(parts["exp"]):
                return False # Privilege has expired

            # Verify node binding identity
            if parts["node"] != expected_node_id:
                return False # Token reuse/spoofing prevention

            # Check explicit capability presence
            granted_capabilities = parts["caps"].split(",")
            return required_capability in granted_capabilities

        except (ValueError, KeyError):
            return False # Malformed token structure
